SELECT
    ot.id AS ordre_id,
    c.nom_entreprise AS client,
    ch.nom_chantier AS chantier,
    m.nom AS materiel,
    m.poids_kg,
    ot.camion_id,
    ca.modele || ' (' || ca.immatriculation || ')' AS camion_assigne,
    ot.type_action,
    ot.date_planifiee,
    ot.ordre_passage,
    ot.tournee_id,
    ot.statut
FROM ordres_transport ot
LEFT JOIN chantiers ch ON ot.chantier_id = ch.id
LEFT JOIN clients c ON ch.client_id = c.id
LEFT JOIN materiels m ON ot.materiel_id = m.id
LEFT JOIN camions ca ON ot.camion_id = ca.id
WHERE ot.statut = 'planifie'
ORDER BY ot.date_planifiee DESC, ot.camion_id, ot.ordre_passage;
